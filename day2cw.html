// Store customer details const customerName = "Riya"; const customerComment = "The coffee was amazing and the service was excellent!"; // Create review object const review = { name: customerName, comment: customerComment }; // Function to display thank-you message function showReview(reviewObj) { // Convert name to uppercase const upperName = reviewObj.name.toUpperCase(); // Get first 20 characters of comment const shortComment = reviewObj.comment.substring(0, 20); console.log(`Thank you, ${upperName}!`); console.log(`Your review: "${shortComment}..."`); } showReview(review);
